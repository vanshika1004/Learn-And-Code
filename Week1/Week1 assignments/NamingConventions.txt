Assignment 2: The below program is to guess the correct number between 1 to 100

def isValidNumber(guessInput):
    if guessInput.isdigit() and 1<= int(guessInput) <=100:
        return True
    else:
        return False

def main():
    targetNumber=random.randint(1,100)
    guessedCorrectly=False
    currentGuess=input("Guess a number between 1 and 100:")
    numberOfGuesses=0
    while not guessedCorrectly:
        if not isValidNumber(currentGuess):
            currentGuess=input("I wont count this one Please enter a number between 1 to 100")
            continue
        else:
            numberOfGuesses+=1
            currentGuess=int(currentGuess)

        if currentGuess<targetNumber:
            currentGuess=input("Too low. Guess again")
        elif currentGuess>targetNumber:
            currentGuess=input("Too High. Guess again")
        else:
            print("You guessed it in",numberOfGuesses,"guesses!")
            guessesCorrectly=True


main()

Redundant Comment - These comments state the obvious wriiten in the code.
Noise Comment - These comments provide no new insight.
Misleading Comment - These comments contradict the code, are no longer accurate, or suggest behavior 
that the code does not actually perform.
ToDo Comment - TODO comments indicate tasks that still need to be done, problems that must be fixed, 
or code that is incomplete.
Journal Comment - Journal comments are comments that act like a change log, recording updates, authors, 
or dates directly in the source code.
ByLine comment - Byline comments are comments that state who wrote the code.

I learned that unnecessary comments should be avoided because well-written, readable code can explain its own
intent.Clear method names, meaningful variable names, and proper structure reduce the need for comments. Comments
should not be used to explain obvious logic, but only where the intent, business rules, or consequences are not 
immediately clear from the code. The goal is to write code that communicates clearly first and rely on comments
only when they add real value.

public class OrderProcessor
{
    private readonly IPaymentGateway _paymentGateway;
    private readonly IInventoryService _inventoryService;
    private readonly INotificationService _notificationService;

    public OrderProcessor(
        IPaymentGateway paymentGateway,
        IInventoryService inventoryService,
        INotificationService notificationService)
    {
        _paymentGateway = paymentGateway;
        _inventoryService = inventoryService;
        _notificationService = notificationService;
    }

    public async Task<OrderResult> ProcessOrder(Order order)
    {
        if (order == null)
        {
            throw new ArgumentNullException(nameof(order));
        }

        if (!IsValidOrder(order))
        {
            return OrderResult.Invalid("Order validation failed");
        }

        bool hasInventory = await _inventoryService.CheckAvailability(order.Items);

        if (!hasInventory)
        {
            return OrderResult.Failed("Insufficient inventory");
        }

        await _inventoryService.ReserveItems(order.Items);

        try
        {
            var paymentResult =
                await _paymentGateway.ProcessPayment(
                    order.CustomerId,
                    order.TotalAmount,
                    order.PaymentMethod);

            if (paymentResult.IsSuccessful)
            {
                await _inventoryService.CommitReservation(order.Items);
                await _notificationService.SendOrderConfirmation(order);

                return OrderResult.Success(paymentResult.TransactionId);
            }
            else
            {
                await _inventoryService.ReleaseReservation(order.Items);
                return OrderResult.Failed($"Payment failed: {paymentResult.ErrorMessage}");
            }
        }
        catch (Exception ex)
        {
            await _inventoryService.ReleaseReservation(order.Items);
            Console.WriteLine($"Error processing order: {ex.Message}");
            throw;
        }
    }

    private bool IsValidOrder(Order order)
    {

    }

    public async Task CancelOrder(string orderId)
    {
        var order = await GetOrderById(orderId);

        if (order.Status == OrderStatus.Paid)
        {
            await _paymentGateway.RefundPayment(order.TransactionId);
            await _inventoryService.RestoreInventory(order.Items);
        }

        order.Status = OrderStatus.Cancelled;
        await SaveOrder(order);
    }

}
